# Guess_Constellations - runtime dependencies
# Verified working on Python 3.13.5 (CPython), Windows/Linux desktop.
#
# Install:
#   python -m pip install -r requirements.txt
#
# Run:
#   python main.py                 # normal quiz
#   python main.py -t              # verify all stimulus images load
#   python main.py -i              # (flag exists but is currently a no-op)
#   python main.py -d 0.6          # target difficulty sum for the 3 drawn items

# --- Transitive dependencies (pulled in by customtkinter; pinned for reproducibility) ---
darkdetect==0.8.0        # OS light/dark theme detection used by customtkinter
packaging==25.0          # version parsing used by customtkinter

# --- Not installable from PyPI ---

#   Fedora:         sudo dnf install python3-tkinter
#   macOS/Windows:  included with the python.org installer
#
